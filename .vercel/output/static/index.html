from fastapi import FastAPI, Depends
from bedrock_agent import BedrockClient

app = FastAPI(title="HRMS AI Engine")

@app.get("/api/v1/analytics")
async def get_summary(
    dept_id: str, 
    client: BedrockClient = Depends()
):
    data = await client.query(dept_id)
    return {"status": 200, "metrics": data}
